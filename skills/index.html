    TCL — EDA Automation
    # Example: Innovus batch mode frequency sweep
set freqs {500 750 1000 1250 1500 2000}
foreach freq $freqs {
    set period [expr {1000.0 / $freq}]
    create_clock -name clk -period $period [get_ports clk]
    place_design
    ccopt_design
    route_design
    report_timing > reports/timing_${freq}MHz.rpt
}
